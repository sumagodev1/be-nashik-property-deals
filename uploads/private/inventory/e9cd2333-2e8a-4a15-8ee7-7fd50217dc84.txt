import streamlit as st

# ================= Config =================
DEFAULT_OLLAMA_MODEL = "llama3.2:latest"  # make sure this model is pulled in Ollama
MAX_HISTORY_TURNS = 20
# ==========================================

# Optional: local Ollama client
try:
    import ollama  # type: ignore
    HAVE_OLLAMA = True
except Exception:
    HAVE_OLLAMA = False


def ollama_chat(model: str, system: str, user: str, temperature: float = 0.1, num_predict: int = 512) -> str:
    """Small helper around ollama.chat. Returns empty string on error."""
    if not HAVE_OLLAMA:
        return ""
    try:
        res = ollama.chat(  # type: ignore[name-defined]
            model=model,
            messages=[
                {"role": "system", "content": system},
                {"role": "user", "content": user},
            ],
            options={"temperature": temperature, "num_ctx": 4096, "num_predict": num_predict},
        )
        return (res.get("message", {}) or {}).get("content", "").strip()
    except Exception:
        return ""


def build_answer(question: str, script: str, model: str) -> str:
    """Build the next reply in a natural, script-grounded conversation.

    Rules (for the model, not to be spoken aloud):
    - You are *Pooja* (or the agent defined in the script), speaking on a phone call.
    - Always stay inside this call context; do not act like an AI model.
    - Never mention words like "script", "guidelines", "conversation history", or describe your reasoning.
    - Speak in short, natural sentences as if you are talking to the user on the phone.
    - Use the user's name only if it already appears in the conversation; otherwise, avoid guessing.
    - If the user asks something off-topic, you may answer briefly, but you must quickly connect back
      to the call goal from the script and keep the flow on track.
    """
    if not HAVE_OLLAMA:
        return (
            "Ollama Python client or server is not available. "
            "Start `ollama serve` and ensure the Python package is installed."
        )

    # Build a compact text version of the recent conversation for context
    history = st.session_state.get("messages", [])[-MAX_HISTORY_TURNS * 2 :]
    convo_lines = []
    for msg in history:
        role = msg.get("role")
        content = (msg.get("content") or "").strip()
        if not content:
            continue
        if role == "user":
            convo_lines.append(f"User: {content}")
        else:  # assistant / bot
            convo_lines.append(f"Pooja: {content}")
    convo_text = "\n".join(convo_lines)

    system = (
        "You are Pooja, a human call agent following the company call script. "
        "You are on a live phone call with the user.\n\n"
        "Speak in simple, natural, human language. No stage directions, no bullet points, no brackets, "
        "no explanations about being an AI or about scripts. Just say what you would say next on the call.\n\n"
        "Very important conversation rules (follow silently):\n"
        "- Never correct the user's grammar or spelling. Do NOT say things like 'I think you meant to say...'.\n"
        "- Never say 'let's start fresh' or ask the user to repeat the same sentence.\n"
        "- Do not repeat the exact same question many times; if they seem unsure, gently offer 1-2 suggestions instead.\n"
        "- If the user says things like 'you suggest' or 'you suggest me', treat it as a request for your suggestion and\n"
        "  propose a few concrete IT solution options, then ask one short follow-up question to move the call forward.\n"
        "- Keep replies short (1-3 sentences) and always stay friendly and confident.\n\n"
        "If the user asks something unrelated (for example, about your identity, name source, or other topics),\n"
        "answer briefly in one short sentence and then smoothly bring the conversation back to the purpose of the call,\n"
        "using the script as your guide. Do not over-explain. One concise reply per turn."
    )

    user = (
        "CALL SCRIPT (for your reference only, never mention this explicitly):\n" + (script or "(no script provided)") + "\n\n"
        "Conversation so far (User = customer, Pooja = you):\n" + (convo_text or "(no previous messages)") + "\n\n"
        "The user just said: " + question.strip() + "\n\n"
        "Reply with only what Pooja says next in this call."
    )

    answer = ollama_chat(model, system, user, temperature=0.2, num_predict=256)
    if not answer:
        answer = (
            "I could not generate a response. Please check that Ollama is running "
            "and the model '" + model + "' is available."
        )
    return answer


# --------------- Streamlit UI ----------------
st.set_page_config(page_title="Script Q&A Chatbot", layout="centered")
st.title("Script-based Q&A Chatbot (Ollama)")

# --- Session state ---
if "messages" not in st.session_state:
    st.session_state.messages = []  # list[dict(role, content)]
if "script_text" not in st.session_state:
    st.session_state.script_text = ""
if "model_tag" not in st.session_state:
    st.session_state.model_tag = DEFAULT_OLLAMA_MODEL


# --- Sidebar: configuration & script loading ---
with st.sidebar:
    st.header("Configuration")
    st.session_state.model_tag = st.text_input(
        "Ollama model tag",
        value=st.session_state.model_tag,
        help="Example: llama3.2:latest or llama3.2:3b-instruct",
    )

    if not HAVE_OLLAMA:
        st.warning("Ollama client is not available. Install it and run `ollama serve`.")

    st.subheader("Load script")

    # Directly paste or edit the full script here. No file upload needed.
    script_text_area = st.text_area(
        "Or paste / edit script text",
        value=st.session_state.script_text,
        height=220,
    )
    if st.button("Use pasted text as script"):
        st.session_state.script_text = script_text_area

    if st.session_state.script_text:
        word_count = len(st.session_state.script_text.split())
        st.caption(f"Current script length: {word_count} words")

st.write("Load your script from the sidebar, then chat below.")

# Optional preview of the current script
if st.session_state.script_text:
    with st.expander("Current script (preview)", expanded=False):
        # Show at most first ~2000 characters to avoid huge UI blocks
        preview = st.session_state.script_text[:2000]
        if len(st.session_state.script_text) > 2000:
            preview += "\n... (truncated)"
        st.text(preview)
else:
    st.info("No script loaded yet. You can still ask general questions (they will be answered using Ollama).")

# --- Render chat history ---
for msg in st.session_state.messages[-MAX_HISTORY_TURNS * 2 :]:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# --- Chat input ---
user_input = st.chat_input("Ask a question about the script or anything else…")

if user_input:
    # Show user message
    st.session_state.messages.append({"role": "user", "content": user_input})
    with st.chat_message("user"):
        st.markdown(user_input)

    # Build answer via Ollama (script-aware)
    answer_text = build_answer(user_input, st.session_state.script_text, st.session_state.model_tag)

    st.session_state.messages.append({"role": "assistant", "content": answer_text})
    with st.chat_message("assistant"):
        st.markdown(answer_text)
